## 使用

```ts
import { createApp } from '@zhengxs/bottle';

const app = createApp();

app.use(function () {
  ctx.statusCode = 200;
  ctx.body = 'hello,world';
});

app.listen(3000, function () {
  console.log('Server is running on http://localhost:3000');
});
```
